"""Owned-lab department configuration: test before implementation."""
import copy
import importlib.util
from pathlib import Path
import unittest
import uuid

ROOT=Path(__file__).resolve().parents[1]
SOURCE=ROOT/'deploy/linux/department_workflow.py'
workflow=None
if SOURCE.exists():
    spec=importlib.util.spec_from_file_location('department_workflow',SOURCE)
    workflow=importlib.util.module_from_spec(spec);spec.loader.exec_module(workflow)

def ident(value):return str(uuid.UUID(int=value))

def fixture():
    instance=ident(100)
    identity={'instance_id':instance,'address':'192.0.2.41'}
    ids={'organization':ident(1),'warehouse':ident(2),'owner':ident(3),'actor_biuro':ident(4),'actor_magazynier':ident(5)}
    users={'schema':1,'instance_id':instance,'origin':'https://192.0.2.41','ids':ids,
           'users':{'biuro':{'username':'biuro','subject':ident(6)},'magazynier':{'username':'magazynier','subject':ident(7)}}}
    positions=[{'id':ident(20+i),'organization_id':ids['organization'],'warehouse_id':ids['warehouse'],'code':code,'kind':'storage','name':'Regał '+code} for i,code in enumerate(['A1','A2','A3','A4','B1','B2','B3','B4'])]
    workspace={'organizations':[{'id':ids['organization'],'name':'METALFORM'}],
               'warehouses':[{'id':ids['warehouse'],'organization_id':ids['organization'],'name':'METALFORM · magazyn'}],
               'owners':[{'id':ids['owner'],'organization_id':ids['organization'],'name':'Towar własny'}],
               'items':[{'id':ident(9),'organization_id':ids['organization'],'name':'Śruba M8 × 40','sku':'SR-M8X40','unit':'szt.','quantity_scale':0}],
               'positions':positions,'recipes':[],'future_metadata':{'preserve':True}}
    return identity,users,workspace

class DepartmentWorkflowTests(unittest.TestCase):
    def setUp(self):
        self.assertIsNotNone(workflow,'The department workflow configurator must exist before the lab can safely enable authenticated transfers')
        self.identity,self.users,self.workspace=fixture()

    def plan(self):return workflow.build_plan(self.identity,self.users,self.workspace,self.identity['instance_id'])

    def test_exact_route_and_named_employees(self):
        plan=self.plan()
        self.assertEqual([d['name'] for d in plan['departments']],['Magazyn','Spawanie','Szlifowanie','Lakiernia','Montaż'])
        by_id={d['id']:d['code'] for d in plan['departments']}
        edges=[(by_id[r['source_department_id']],by_id[r['destination_department_id']]) for r in plan['routes']]
        self.assertEqual(edges,[('magazyn','spawanie'),('spawanie','szlifowanie'),('szlifowanie','lakiernia'),('lakiernia','montaz'),('montaz','magazyn')])
        self.assertEqual({p['username'] for p in plan['employees']},{'jan.kowalski','anna.zielinska','piotr.nowak','marek.wisniewski'})
        self.assertEqual(len({p['actor_id'] for p in plan['employees']}),4)
        self.assertTrue(all(p['firstName'] and p['lastName'] and p['display_name'] for p in plan['employees']))

    def test_deterministic_without_mutating_existing_configuration(self):
        before=copy.deepcopy((self.identity,self.users,self.workspace))
        self.assertEqual(self.plan(),self.plan())
        self.assertEqual((self.identity,self.users,self.workspace),before)

    def test_instances_use_independent_department_actor_and_position_ids(self):
        first=self.plan();other=copy.deepcopy(self.identity);other['instance_id']=ident(101)
        users=copy.deepcopy(self.users);users['instance_id']=other['instance_id']
        second=workflow.build_plan(other,users,self.workspace,other['instance_id'])
        for group in ['departments','employees','positions']:
            key='actor_id' if group=='employees' else 'id'
            self.assertTrue({r[key] for r in first[group]}.isdisjoint({r[key] for r in second[group]}),group)

    def test_scope_drift_and_wrong_installation_fail_closed(self):
        with self.assertRaises(ValueError):workflow.build_plan(self.identity,self.users,self.workspace,ident(999))
        users=copy.deepcopy(self.users);users['instance_id']=ident(999)
        with self.assertRaises(ValueError):workflow.build_plan(self.identity,users,self.workspace,self.identity['instance_id'])
        workspace=copy.deepcopy(self.workspace);workspace['positions'][0]['warehouse_id']=ident(999)
        with self.assertRaises(ValueError):workflow.build_plan(self.identity,self.users,workspace,self.identity['instance_id'])

    def test_missing_or_ambiguous_rack_address_refused(self):
        for positions in [self.workspace['positions'][1:],self.workspace['positions']+[copy.deepcopy(self.workspace['positions'][0])]]:
            workspace=copy.deepcopy(self.workspace);workspace['positions']=positions
            with self.assertRaises(ValueError):workflow.build_plan(self.identity,self.users,workspace,self.identity['instance_id'])

    def test_merge_only_adds_four_places_preserving_items_and_unknown_metadata(self):
        plan=self.plan();before=copy.deepcopy(self.workspace)
        merged=workflow.merge_workspace(self.workspace,plan)
        self.assertEqual(self.workspace,before)
        self.assertEqual(merged['items'],before['items'])
        self.assertEqual(merged['future_metadata'],before['future_metadata'])
        self.assertEqual(len(merged['positions']),len(before['positions'])+4)
        self.assertEqual(workflow.merge_workspace(merged,plan),merged)
        self.assertNotIn('in_transit',[p['code'] for p in plan['positions']],'Transit places belong to individual commands, not setup')

    def test_merge_refuses_changed_preexisting_department_place(self):
        plan=self.plan();merged=workflow.merge_workspace(self.workspace,plan)
        merged['positions'][-1]['name']='Zmiana bez zgody'
        with self.assertRaises(ValueError):workflow.merge_workspace(merged,plan)

    def test_operators_have_no_generic_posting_or_document_creation_grants(self):
        plan=self.plan();employee_ids={p['actor_id'] for p in plan['employees']}
        grants=[g for g in plan['grants'] if g['actor_id'] in employee_ids]
        self.assertEqual({g['permission'] for g in grants},{'document.read','stock.read','handoff.read','handoff.send','handoff.accept'})
        self.assertTrue(all(g['organization_id']==self.users['ids']['organization'] and g['warehouse_id']==self.users['ids']['warehouse'] and g['owner_id']==self.users['ids']['owner'] for g in plan['grants']))
        bureau={g['permission'] for g in plan['grants'] if g['actor_id']==self.users['ids']['actor_biuro']}
        self.assertEqual(bureau,{'handoff.read'})

    def test_database_seed_only_configures_metadata_and_grants(self):
        plan=self.plan();subjects={p['username']:ident(200+i) for i,p in enumerate(plan['employees'])}
        sql=workflow.seed_sql(plan,subjects)
        for table in ['warehouse.departments','warehouse.department_positions','warehouse.department_employees','warehouse.department_members','warehouse.department_routes','platform.actor_grants']:
            self.assertIn(table,sql)
        for forbidden in ['post_movement','on_hand','reserved_qty','warehouse.documents','warehouse.work_orders','wms_migrations','TRUNCATE','DELETE FROM']:
            self.assertNotIn(forbidden,sql)
        self.assertIn('ON CONFLICT',sql)

    def test_seed_rejects_missing_or_injected_identity_subjects(self):
        plan=self.plan()
        with self.assertRaises(ValueError):workflow.seed_sql(plan,{})
        subjects={p['username']:ident(200+i) for i,p in enumerate(plan['employees'])}
        subjects['jan.kowalski']="'; DROP TABLE platform.actors; --"
        with self.assertRaises(ValueError):workflow.seed_sql(plan,subjects)

if __name__=='__main__':unittest.main()
