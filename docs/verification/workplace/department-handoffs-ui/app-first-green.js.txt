async function list(resource,cursor) {
  const params = new URLSearchParams({...S.scope,limit:'100'});
  if (cursor) params.set('cursor',cursor);
  return request(`/api/v1/${resource}?${params}`);
}
async function refresh(append=false) {
  const page = await list('documents',append?S.cursor:null);
  S.documents = append ? [...S.documents,...page.data.filter(row => !S.documents.some(old => old.id === row.id))] : page.data;
  S.cursor = page.pagination?.next_cursor;
  if (can('stock.read')) {
    const balances = [], cursors = new Set(); let cursor;
    do {
      const stock = await list('stock-balances',cursor);
      balances.push(...stock.data.filter(row => !balances.some(old => old.id === row.id)));
      cursor = stock.pagination?.next_cursor;
      if (cursor && cursors.has(cursor)) throw new Error('Nie udało się wczytać wszystkich stanów. Odśwież magazyn.');
      if (cursor) cursors.add(cursor);
    } while (cursor);
    S.balances = balances; S.stockCursor = null;
  }
  const orders = await list('work-orders'); S.orders = orders.data; S.orderCursor = orders.pagination?.next_cursor;
  if (can('handoff.read')) {
    S.handoffContext=(await request(`/api/v1/handoffs/context?${query()}`)).data;
    const handoffs=await list('handoffs');S.handoffs=handoffs.data;S.handoffCursor=handoffs.pagination?.next_cursor;
    if (S.handoffID) S.handoff=(await request(`/api/v1/handoffs/${S.handoffID}?${query()}`)).data;
  }
  if (S.selected) S.selected = (await request(`/api/v1/documents/${S.selected.id}?${query()}`)).data;
}

async function selectHandoff(id) {
  if (selectionBlocked()) return;
  S.handoffLoading=true;S.handoffID=id;render();

    const record=(await request(`/api/v1/handoffs/${id}?${query()}`)).data;
    if (S.handoffID !== id) return;

    sessionStorage.setItem(`wms-handoff:${workspace.actor_id}`,id);
    sessionStorage.setItem(`wms-handoff-view:${workspace.actor_id}`,'1');
  } finally {S.handoffLoading=false;render();}

  dialog.querySelector('[data-close-handoff]').onclick=()=>dialog.close();
  dialog.querySelector('#handoff-form').onsubmit=async event=>{
    event.preventDefault();if(selectionBlocked()) return;
    const destinationID=dialog.querySelector('#handoff-destination').value,quantity=dialog.querySelector('#handoff-quantity').value;
    const fieldset=dialog.querySelector('fieldset');S.handoffSubmitting=true;fieldset.disabled=true;
    try {
      const context=(await request(`/api/v1/handoffs/context?${query()}`)).data;
      const params=new URLSearchParams({...S.scope,owner_id:balance.owner_id,item_id:balance.item_id,position_id:balance.position_id,limit:'200'});
      const page=await request(`/api/v1/stock-balances?${params}`);
      const current=page.data.filter(row=>row.id === balanceID);
      if(current.length !== 1 || page.pagination?.next_cursor) throw new Error('Stan towaru zmienił się. Odśwież stany i ponownie wybierz pozycję.');
      const body=handoffSendIntent({context,balance:current[0],item,scope:S.scope,destinationID,quantity});

      await write('/api/v1/handoffs/send',body,'Materiał wysłany. Dział docelowy musi potwierdzić przyjęcie.');
    } catch(error) {
      if(dialog.open) {dialog.querySelector('#handoff-form-error').textContent=problem(error);dialog.querySelector('#handoff-quantity').setAttribute('aria-invalid','true');}
      else {S.error=problem(error);render();}
    } finally {S.handoffSubmitting=false;if(dialog.open){fieldset.disabled=false;dialog.querySelector('#handoff-quantity').focus();}else render();}
  };
  dialog.showModal();dialog.querySelector('#handoff-destination').focus();

async function acceptHandoff() {
  if(selectionBlocked() || !S.handoff || !can('handoff.accept',S.handoff.owner_id)) return;
  const captured=S.handoff;S.handoffSubmitting=true;render();
  try {
    const context=(await request(`/api/v1/handoffs/context?${query()}`)).data;
    const current=(await request(`/api/v1/handoffs/${captured.id}?${query()}`)).data;
    S.handoffContext=context;S.handoff=current;
    if(current.version !== captured.version || current.state !== captured.state) throw new Error('Przekazanie zmieniło się. Sprawdź aktualny stan i potwierdzenie przyjęcia.');
    const body=handoffAcceptIntent({context,handoff:current,scope:S.scope});
    await write(`/api/v1/handoffs/${current.id}/accept`,body,'Przyjęcie materiału potwierdzone na Twoim koncie.');
  } finally {S.handoffSubmitting=false;render();}
}

async function selectionLineChanged(id) {
  const selection={lineID:id,sourceID:'',targetID:'',quantity:'',reservations:[],loading:S.selected.kind === 'shipment' && Boolean(id)};
  S.selection=selection; S.error=''; renderSelectionFocus('selection-line');
  if (!selection.loading) return;
  try {
    const params=new URLSearchParams({...S.scope,document_line_id:id,limit:'200'});
    const page=await request(`/api/v1/reservations?${params}`);
    if (page.pagination?.next_cursor) throw new Error('Pozycja ma wiele rezerwacji. Poproś kierownika o ich sprawdzenie.');
    if (S.selection === selection) selection.reservations=page.data;
  } finally {if (S.selection === selection) {selection.loading=false;renderSelectionFocus('selection-line');}}
}
async function confirmSelection() {
  if (selectionBlocked()) return;
  const selected={...S.selection}, documentID=S.selected.id, version=S.selected.version;
  S.selectSubmitting=true; render();
  try {
    const current=(await request(`/api/v1/documents/${documentID}?${query()}`)).data;
    if (current.version !== version || current.id !== documentID) {
      S.selected=current; resetWork();
      throw new Error('Dokument zmienił się. Sprawdź aktualne ilości i ponownie wybierz pozycję.');
    }
    const stock=[],cursors=new Set();let cursor;
    do {
      const params=new URLSearchParams({...S.scope,owner_id:current.owner_id,limit:'200'});
      if (cursor) params.set('cursor',cursor);
      const page=await request(`/api/v1/stock-balances?${params}`);
      stock.push(...page.data);cursor=page.pagination?.next_cursor;
      if (cursor && cursors.has(cursor)) throw new Error('Nie udało się sprawdzić aktualnego stanu. Odśwież magazyn.');
      if (cursor) cursors.add(cursor);
    } while (cursor);
    let reservations=[];
    if (current.kind === 'shipment') {
      const params=new URLSearchParams({...S.scope,document_line_id:selected.lineID,limit:'200'});
      const page=await request(`/api/v1/reservations?${params}`);
      if (page.pagination?.next_cursor) throw new Error('Pozycja ma wiele rezerwacji. Poproś kierownika o ich sprawdzenie.');
      reservations=page.data;
    }
    const work=resolveSelection({document:current,lineID:selected.lineID,items:workspace.catalog.items,balances:stock,positions:workspace.catalog.positions,scope:S.scope,sourceID:selected.sourceID,targetID:selected.targetID,reservations});
    S.selected=current;
    S.balances=[...S.balances.filter(row=>row.owner_id !== current.owner_id),...stock];
    if (current.kind === 'shipment') await issueShipment(work,selected.quantity);
    else {
      const body=movementIntent({...work,document:current,scope:S.scope,quantity:selected.quantity,kind:current.kind});
      await write('/api/v1/movements',body,current.kind === 'receipt'?'Towar przyjęty':'Towar przesunięty');
    }
  } finally {S.selectSubmitting=false;render();}
}

}

async function run(fn) {
  if (S.busy) return;
  S.error = ''; S.message = '';
  try {await fn();} catch (err) {S.error = problem(err); render(); focusScan();}
}
function problem(err) {
  return errorMessage(err,Boolean(pending()));
}
async function write(path,body,message) {
  if (S.busy) return;
  if (S.needsRefresh) throw new Error('Najpierw odśwież dokument po potwierdzonej operacji');
  S.busy = true; render();
  try {
    const send = () => journal.submit(path,body);
    const result = navigator.locks?await navigator.locks.request('wms:'+journal.namespace,send):await send();
    await afterConfirmation(result,message);
    return result;
  } finally {S.busy = false; render(); focusScan();}
}
async function afterConfirmation(result,message) {
  S.message = message;
  if (result.data.handoff_id) {
    S.handoffID=result.data.handoff_id;S.section='handoffs';S.selected=null;
    sessionStorage.setItem(`wms-handoff:${workspace.actor_id}`,S.handoffID);
    sessionStorage.setItem(`wms-handoff-view:${workspace.actor_id}`,'1');
  }
  try {
    await refresh();
    if (result.data.document_id && S.selected?.id !== result.data.document_id) S.selected = (await request(`/api/v1/documents/${result.data.document_id}?${query()}`)).data;
    if (S.selected) {sessionStorage.setItem(`wms-open:${workspace.actor_id}`,S.selected.id); S.section = S.selected.kind;}
    if (result.data.id && result.data.state === 'planned') {S.order = result.data.id; S.section = 'orders';}
    resetWork(); S.error = ''; S.needsRefresh = false;
  } catch {
    S.needsRefresh = true;
    S.error = 'Operacja potwierdzona. Nie udało się odświeżyć dokumentu. Wybierz „Odśwież”.';
  }
}
function action(name) {
  if (name === 'create'||name === 'putaway') return openCreate(name === 'putaway');
  if (name === 'create-order') return openOrder();
  if (name === 'order-materials'||name === 'order-goods') {
    const order = S.orders.find(row => row.id === S.order);
    if (order) return openCreate(false,{kind:'shipment',owner_id:order.owner_id,lines:name === 'order-materials'?order.requirements:order.lines,recipient:name === 'order-materials'?'Produkcja':order.customer,order_reference:order.reference,sourceOrder:order});
    return;
  }
  if (name === 'terminal') {S.mode = 'worker'; resetWork(); render(); focusScan(); return;}
  if (name === 'deselect'||name === 'back-tasks') {S.selected = null; S.work = null; render(); return;}
  if (name === 'restart-scan') {resetWork(); S.error = ''; render(); focusScan(); return;}
  if (name === 'logout') {identity.logout({redirectUri:location.origin+'/app/'}); return;}
  run(async () => {
    if (name === 'accept-handoff') await acceptHandoff();
    if (name === 'more-handoffs') {const page=await list('handoffs',S.handoffCursor);S.handoffs.push(...page.data.filter(row=>!S.handoffs.some(old=>old.id === row.id)));S.handoffCursor=page.pagination?.next_cursor;render();}
    if (name === 'refresh'||name === 'more') {await refresh(name === 'more'); S.needsRefresh = false; resetWork(); S.message = 'Dane odświeżone'; render();}
    if (name === 'more-stock') {const page = await list('stock-balances',S.stockCursor); S.balances.push(...page.data); S.stockCursor = page.pagination?.next_cursor; render();}
    if (name === 'more-orders') {const page = await list('work-orders',S.orderCursor); S.orders.push(...page.data.filter(row => !S.orders.some(old => old.id === row.id))); S.orderCursor = page.pagination?.next_cursor; render();}
    if (name === 'activate'||name === 'close-document') {const doc = S.selected; await write(`/api/v1/documents/${doc.id}/transitions`,{...S.scope,expected_document_version:doc.version,state:name === 'activate'?'active':'closed'},name === 'activate'?'Dokument przekazany do realizacji':'Dokument zrealizowany');}
    if (name === 'retry') {
      S.busy = true; render();
      try {const result = await journal.retry(); await afterConfirmation(result,'Operacja potwierdzona. Możesz kontynuować pracę.');}
      finally {S.busy = false; render(); focusScan();}
    }
  });
}
async function scanInput(value) {
  if (S.needsRefresh) throw new Error('Najpierw odśwież dokument po potwierdzonej operacji');
  if (S.scanning) return;
  S.scanning = true;
  try {await advanceScan(value);} finally {S.scanning = false;}
}
async function advanceScan(value) {
  const work = S.work;
  if (work.stage === 'item') {matchScan(value,[work.item]); work.stage = S.selected.kind === 'receipt'?'target':'source';}
  else if (work.stage === 'source'||work.stage === 'target') {
    const position = matchScan(value,workspace.catalog.positions.filter(row => row.organization_id === S.scope.organization_id&&row.warehouse_id === S.scope.warehouse_id),'code');
    if (work.stage === 'target' && S.selected.kind === 'receipt' && position.kind !== 'receiving') throw new Error('Wybierz adres strefy przyjęć');
    if (work.stage === 'target' && S.selected.kind === 'transfer' && position.kind !== 'storage') throw new Error('Wybierz adres składowania');
    const params = new URLSearchParams({...S.scope,item_id:work.item.id,position_id:position.id,owner_id:S.selected.owner_id,limit:'200'});
    const page = await request(`/api/v1/stock-balances?${params}`);
    const balances = page.data.filter(row => row.item_id === work.item.id&&row.position_id === position.id&&row.owner_id === S.selected.owner_id);
    if (balances.length !== 1||page.pagination?.next_cursor) throw new Error(balances.length?'Pod adresem jest kilka partii lub stanów towaru. Poproś kierownika o wskazanie właściwej pozycji.':'Towar nie jest przypisany do tego adresu. Sprawdź adres lub skontaktuj się z kierownikiem.');
    work[work.stage === 'source'?'sourceBalance':'targetBalance'] = balances[0];
    if (work.stage === 'source' && S.selected.kind === 'shipment') {
      const reservations = await shipmentReservations(work.line,work.sourceBalance);
      if (reservations.length > 1) throw new Error('Pod adresem jest kilka aktywnych rezerwacji tej pozycji. Poproś kierownika o wskazanie rezerwacji.');
      work.reservationQuantity = reservations.length ? parseQuantity(qty(reservations[0].active_qty),work.item.quantity_scale) : null;
    }
    work.stage = work.stage === 'source'&&S.selected.kind !== 'shipment'?'target':'quantity';
  } else {
    if (S.selected.kind === 'shipment') {await issueShipment(work,value); return;}
    const body = movementIntent({document:S.selected,line:work.line,item:work.item,sourceBalance:work.sourceBalance,targetBalance:work.targetBalance,scope:S.scope,quantity:value,kind:S.selected.kind});
    await write('/api/v1/movements',body,S.selected.kind === 'receipt'?'Towar przyjęty':'Towar przesunięty'); return;
  }
  S.error = ''; render(); focusScan();
}
async function shipmentReservations(line,sourceBalance) {
  const params = new URLSearchParams({...S.scope,document_line_id:line.id,limit:'200'});
  const page = await request(`/api/v1/reservations?${params}`);
  if (page.pagination?.next_cursor) throw new Error('Pozycja ma wiele rezerwacji. Poproś kierownika o wskazanie rezerwacji.');
  return page.data.filter(row => row.document_line_id === line.id && row.balance_id === sourceBalance.id && qty(row.active_qty) !== '0');
}

async function issueShipment(work,value) {
  const documentID = S.selected.id;
  const quantity = parseQuantity(value,work.item.quantity_scale);
  let active = await shipmentReservations(work.line,work.sourceBalance);
  if (!active.length) {
    const body = reservationIntent({document:S.selected,line:work.line,item:work.item,sourceBalance:work.sourceBalance,quantity,scope:S.scope});
    await write('/api/v1/reservations',body,'Towar zarezerwowany do wydania');
    if (S.needsRefresh) return;
    active = await shipmentReservations(work.line,work.sourceBalance);
  }
  if (active.length !== 1) throw new Error('Nie można jednoznacznie wybrać rezerwacji. Odśwież dokument i sprawdź adres.');
  try {remaining(active[0].active_qty,quantity);} catch {throw new Error(`Najpierw wydaj z aktywnej rezerwacji: ${qty(active[0].active_qty)} ${work.item.unit} Potem przygotujesz pozostałą ilość dokumentu.`);}
  if (S.selected.id !== documentID) throw new Error('Wybrano inny dokument. Odśwież wydanie.');
  const line = S.selected.lines.find(row => row.id === work.line.id);
  const sourceBalance = S.balances.find(row => row.id === work.sourceBalance.id);
  const body = issueIntent({document:S.selected,line,item:work.item,sourceBalance,reservation:active[0],quantity,scope:S.scope});
  await write('/api/v1/movements',body,'Towar wydany. Sprawdź odbiorcę i zakończ dokument.');
}

function openStockDocument(kind,balanceID) {

  openCreate(false,{kind,owner_id:source.owner_id,lines:[{item_id:source.item_id,quantity:''}],sourceBalance:source});
}

  dialog.innerHTML = `<form id="create-form" novalidate><div class="modal-heading"><div><span class="eyebrow">NOWY DOKUMENT</span><h2 id="create-title">${kind === 'receipt'?'Nowe przyjęcie (PZ)':kind === 'shipment'?'Nowe wydanie (WZ)':'Nowe przesunięcie (MM)'}</h2></div><button type="button" class="icon-button" id="close-dialog" aria-label="Zamknij formularz">${icon('close')}</button></div>${fromReceipt?`<p class="source-note">Na podstawie przyjęcia ${e(S.selected.number)}. Przesunięcie (MM) przeniesie towar ze strefy przyjęć pod adres składowania.</p>`:''}<p class="hint">Numer zostanie nadany automatycznie po zapisaniu</p><div class="form-grid"><div><label for="owner">Właściciel towaru</label><select id="owner" name="owner">${owners.map(owner => `<option value="${e(owner.id)}" ${fromReceipt&&owner.id === S.selected.owner_id?'selected':''}>${e(owner.name)}</option>`).join('')}</select></div></div><div class="detail-section-heading"><h3>Planowane towary</h3><span>do 200 pozycji</span></div><div id="create-lines">${lines.map(row).join('')}</div><button type="button" class="text-button" id="add-line">${icon('plus')} Dodaj towar</button><div id="form-error" role="alert"></div><div class="modal-footer"><p>Zapisz dokument, sprawdź plan i przekaż go do realizacji. Zapis nie zmienia stanów magazynowych.</p><button class="button primary" type="submit">Zapisz dokument ${icon('arrow')}</button></div></form>`;
  dialog.querySelector('#close-dialog').onclick = () => dialog.close();
  if (options.owner_id) dialog.querySelector('[name=owner]').value = options.owner_id;
  if (options.sourceOrder) dialog.querySelector('.hint').insertAdjacentHTML('afterend',`<p class="source-note">Zamówienie: <strong>${e(options.sourceOrder.reference)}</strong><br>Firma: ${e(options.sourceOrder.customer)}</p>`);
  if (kind === 'shipment') dialog.querySelector('.form-grid').insertAdjacentHTML('beforeend',`<div><label for="recipient">Odbiorca</label><input id="recipient" name="recipient" maxlength="200" autocomplete="organization" value="${e(options.recipient||'')}" placeholder="Firma, dział lub osoba odbierająca towar" aria-required="true"></div><div><label for="order-reference">Numer zamówienia</label><input id="order-reference" name="order_reference" maxlength="100" value="${e(options.order_reference||'')}" placeholder="Np. zamówienie klienta #1 (opcjonalnie)"></div>`);
  if (options.sourceBalance) dialog.querySelector('.hint').insertAdjacentHTML('afterend',`<p class="source-note">Z miejsca <strong>${e(positionById(options.sourceBalance.position_id)?.code)}</strong> · dostępne ${e(qty(options.sourceBalance.available_qty))} ${e(itemById(options.sourceBalance.item_id)?.unit)}. Zapis dokumentu nie zmienia stanu. Towar potwierdzisz podczas realizacji.</p>`);
  dialog.onclick = event => {const button = event.target.closest('[data-remove-line]'); if (button&&dialog.querySelectorAll('.create-line').length > 1) button.closest('.create-line').remove();};
  dialog.querySelector('#add-line').onclick = () => {if (dialog.querySelectorAll('.create-line').length < 200) dialog.querySelector('#create-lines').insertAdjacentHTML('beforeend',row({item_id:items[0].id,quantity:''},sequence++));};
  dialog.querySelector('#create-form').onsubmit = event => {
    event.preventDefault();
    try {
      const recipient = dialog.querySelector('[name=recipient]');
      if (kind === 'shipment' && !recipient.value.trim()) {recipient.setAttribute('aria-invalid','true'); recipient.focus(); throw new Error('Wpisz odbiorcę wydania');}
      if (recipient && [...recipient.value.trim()].length > 200) {recipient.setAttribute('aria-invalid','true'); recipient.focus(); throw new Error('Odbiorca może mieć najwyżej 200 znaków');}
      const inputLines = [...dialog.querySelectorAll('.create-line')].map(container => {const item = items.find(item => item.id === container.querySelector('[name=item]').value), field = container.querySelector('[name=quantity]'); try {return {item_id:item.id,quantity:parseQuantity(field.value,item.quantity_scale)};} catch (err) {field.setAttribute('aria-invalid','true'); field.focus(); throw err;}});
      const body = {...S.scope,owner_id:dialog.querySelector('[name=owner]').value,kind,lines:inputLines};
      if (options.sourceBalance) {
        const source = options.sourceBalance;
        if (inputLines.length !== 1 || inputLines[0].item_id !== source.item_id || body.owner_id !== source.owner_id) throw new Error('Operacja z miejsca dotyczy wybranego towaru i właściciela. Otwórz inną pozycję w stanach.');
        if (materialAvailability(inputLines,[source],{...S.scope,owner_id:source.owner_id})[0].shortage_qty !== '0') throw new Error('Pod adresem nie ma wystarczającej ilości dostępnego towaru');
      }
      if (kind === 'shipment') {body.recipient = recipient.value.trim(); const reference = dialog.querySelector('[name=order_reference]').value.trim(); if ([...reference].length > 100) throw new Error('Numer zamówienia może mieć najwyżej 100 znaków'); if (reference) body.order_reference = reference;}
      dialog.close(); S.section = kind; S.filter = 'all'; S.search = '';
      run(() => write('/api/v1/documents',body,'Dokument zapisany. Sprawdź plan i przekaż go do realizacji.'));
    } catch (err) {dialog.querySelector('#form-error').textContent = err.message;}
  };
  dialog.showModal();
}
function openOrder() {
  if (pending() || S.busy || S.needsRefresh || !can('document.create')) return;
  const dialog = root.querySelector('#create-dialog');
  const owners = workspace.catalog.owners.filter(row => row.organization_id === S.scope.organization_id && can('document.create',row.id));
  const items = workspace.catalog.items.filter(row => row.organization_id === S.scope.organization_id && row.archived !== true);
  const recipes = (workspace.catalog.recipes || []).filter(row => row.organization_id === S.scope.organization_id && row.warehouse_id === S.scope.warehouse_id && items.some(item => item.id === row.product_item_id));
  if (!owners.length || !items.length) {S.error = 'Brakuje towarów lub właściciela do zaplanowania zamówienia'; render(); return;}
  let sequence = 0;
  const row = (itemID,quantity='') => {const n = sequence++; return `<div class="create-line"><div><label for="ordered-item-${n}">Zamówiony towar</label><select id="ordered-item-${n}" name="ordered-item">${items.map(item => `<option value="${e(item.id)}" ${item.id === itemID?'selected':''}>${e(item.sku)} · ${e(item.name)}${item.archived === true?' · Wcześniejsze próby':''}</option>`).join('')}</select></div><div><label for="ordered-quantity-${n}">Ilość zamówiona</label><input id="ordered-quantity-${n}" name="ordered-quantity" inputmode="decimal" autocomplete="off" value="${e(quantity)}"></div><button type="button" class="icon-button" data-remove-order-line aria-label="Usuń pozycję zamówienia">${icon('close')}</button></div>`;};
  dialog.innerHTML = `<form id="order-form" novalidate><div class="modal-heading"><div><span class="eyebrow">PLAN DLA BIURA I MAGAZYNU</span><h2 id="create-title">Nowe zamówienie</h2></div><button type="button" class="icon-button" id="close-dialog" aria-label="Zamknij formularz">${icon('close')}</button></div><div class="form-grid"><div><label for="order-customer">Firma lub odbiorca</label><input id="order-customer" name="customer" maxlength="200" placeholder="Np. BIUROMAX Sp. z o.o." aria-required="true"></div><div><label for="customer-reference">Numer zamówienia</label><input id="customer-reference" name="reference" maxlength="100" placeholder="Np. zamówienie klienta #1; puste = numer automatyczny"></div><div><label for="order-owner">Właściciel towaru</label><select id="order-owner" name="owner">${owners.map(owner => `<option value="${e(owner.id)}">${e(owner.name)}</option>`).join('')}</select></div><div><label for="order-recipe">Sposób realizacji</label><select id="order-recipe" name="recipe"><option value="">Wydanie gotowych towarów</option>${recipes.map(recipe => `<option value="${e(recipe.id)}">Produkcja / kompletacja · ${e(recipe.name)}</option>`).join('')}</select></div></div><h3>Zamówione towary</h3><div id="order-lines">${row(items[0].id)}</div><button type="button" class="text-button" id="add-order-line">${icon('plus')} Dodaj towar</button><div id="material-preview" aria-live="polite"></div><div id="form-error" role="alert"></div><div class="modal-footer"><p>Zamówienie jest planem. Nie zmienia stanów i nie potwierdza wykonania produktu. Wydanie przygotujesz osobnym dokumentem WZ.</p><button class="button primary" type="submit">Zapisz zamówienie</button></div></form>`;
  dialog.querySelector('#close-dialog').onclick = () => dialog.close();
  const preview = () => {
    const recipe = recipes.find(row => row.id === dialog.querySelector('[name=recipe]').value);
    const target = dialog.querySelector('#material-preview');
    if (!recipe) {target.innerHTML = '<p class="hint">Magazyn wyda zamówione gotowe towary pod wskazany dokument WZ.</p>'; return;}
    try {
      const product = items.find(row => row.id === recipe.product_item_id);
      const quantity = parseQuantity(dialog.querySelector('[name=ordered-quantity]').value,product.quantity_scale);
      const requirements = materialRequirements(recipe,quantity);
      target.innerHTML = `<h3>Materiały do przygotowania</h3>${materialsTable(requirements,dialog.querySelector('[name=owner]').value)}<h3>Kolejność prac</h3><ol class="production-steps">${recipe.steps.map(step => `<li>${e(step)}</li>`).join('')}</ol><p class="hint">Dostępne ilości dotyczą obecnego stanu; inne zaplanowane zamówienia nie są rezerwacją.</p>`;
    } catch (error) {target.innerHTML = `<p class="hint">${e(error.message)}. Wpisz ilość, aby obliczyć materiały.</p>`;}
  };
  dialog.querySelector('[name=recipe]').onchange = () => {
    const recipe = recipes.find(row => row.id === dialog.querySelector('[name=recipe]').value);
    if (recipe) {
      dialog.querySelector('#order-lines').innerHTML = row(recipe.product_item_id,'');
      dialog.querySelector('[name=ordered-item]').disabled = true;
    } else dialog.querySelector('[name=ordered-item]').disabled = false;
    dialog.querySelector('#add-order-line').hidden = Boolean(recipe); preview();
  };
  dialog.querySelector('#add-order-line').onclick = () => {if (dialog.querySelectorAll('.create-line').length < 200) dialog.querySelector('#order-lines').insertAdjacentHTML('beforeend',row(items[0].id));};
  dialog.onclick = event => {const button = event.target.closest('[data-remove-order-line]'); if (button && dialog.querySelectorAll('.create-line').length > 1) {button.closest('.create-line').remove(); preview();}};
  dialog.querySelector('#order-form').oninput = preview;
  dialog.querySelector('#order-form').onsubmit = event => {
    event.preventDefault();
    try {
      const customer = dialog.querySelector('[name=customer]');
      if (!customer.value.trim()) {customer.focus(); throw new Error('Wpisz firmę lub odbiorcę zamówienia');}
      if ([...customer.value.trim()].length > 200) {customer.focus(); throw new Error('Firma lub odbiorca może mieć najwyżej 200 znaków');}
      const recipe = recipes.find(row => row.id === dialog.querySelector('[name=recipe]').value);
      const lines = [...dialog.querySelectorAll('.create-line')].map(container => {
        const item = items.find(row => row.id === container.querySelector('[name=ordered-item]').value), field = container.querySelector('[name=ordered-quantity]');
        try {return {item_id:item.id,quantity:parseQuantity(field.value,item.quantity_scale)};} catch (error) {field.focus(); throw error;}
      });
      if (recipe && (lines.length !== 1 || lines[0].item_id !== recipe.product_item_id)) throw new Error('Wybierz produkt pasujący do składu');
      if (recipe) materialRequirements(recipe,lines[0].quantity);
      const body = {...S.scope,owner_id:dialog.querySelector('[name=owner]').value,customer:customer.value.trim(),reference:dialog.querySelector('[name=reference]').value.trim(),lines};
      if ([...body.reference].length > 100) throw new Error('Numer zamówienia może mieć najwyżej 100 znaków');
      if (recipe) body.recipe_id = recipe.id;
      dialog.close(); S.section = 'orders';
      run(() => write('/api/v1/work-orders',body,'Zamówienie zapisane. Materiały i zadania są zaplanowane.'));
    } catch (error) {dialog.querySelector('#form-error').textContent = error.message;}
  };
  preview(); dialog.showModal();
}

async function start() {
  try {
    if (S.demo) {const demo = new DemoWarehouse(localStorage); request = demo.request.bind(demo);}
    else {
      const response = await fetch('/app/config.json',{cache:'no-store'});
      if (!response.ok) throw new Error('Nie udało się wczytać ustawień logowania');
      const config = await response.json();
      if (!config.oidc?.url||!config.oidc?.realm||!config.oidc?.clientId) {root.innerHTML = `<main class="login-screen"><div class="login-mark">${icon('box')}</div><span class="eyebrow">STANOWISKO PRACY WMS</span><h1>Twój magazyn.<br>Wszystko na swoim miejscu.</h1><p>Przyjęcia (PZ) i przesunięcia (MM) dla biura i pracowników magazynu.</p><div class="notice">Logowanie nie jest jeszcze skonfigurowane. Administrator musi podłączyć konta pracowników i kartoteki.</div><a class="button primary" href="/app/?demo=1">Otwórz magazyn szkoleniowy ${icon('arrow')}</a><small>Operacje szkoleniowe nie zmieniają stanów magazynu.</small></main>`; return;}
      identity = new Keycloak(config.oidc);
      try {await identity.init({onLoad:'login-required',pkceMethod:'S256',locale:'pl',checkLoginIframe:false,redirectUri:location.origin+'/app/'});}
      catch {throw new Error('Nie udało się zalogować. Sprawdź połączenie i spróbuj ponownie.');}
      request = async (path,body,key) => {
        try {await identity.updateToken(30);} catch {throw Object.assign(new Error('Sesja wygasła. Zaloguj się ponownie.'),{status:401});}
        const headers = {Authorization:'Bearer '+identity.token};
        if (body) {headers['Content-Type']='application/json'; headers['Idempotency-Key']=key;}
        const response = await fetch(path,{method:body?'POST':'GET',headers,body:body?JSON.stringify(body):undefined,cache:'no-store',redirect:'error'});
        const result = await response.json();
        if (!response.ok) throw Object.assign(new Error(result.detail||'Serwer odrzucił operację'),{status:response.status,code:result.code});
        return result;
      };
    }
    workspace = (await request('/api/v1/workspace')).data;
    const first = workspace.scopes.find(scope => scope.grants.some(grant => grant.permission === 'document.read'));
    if (first) {setScope(first); await refresh();}
    const saved = sessionStorage.getItem(`wms-open:${workspace.actor_id}`);
    if (saved&&S.scope) {try {S.selected = (await request(`/api/v1/documents/${saved}?${query()}`)).data; S.section = S.selected.kind;} catch {}}
    if (sessionStorage.getItem(`wms-mode:${workspace.actor_id}`) === 'worker') S.mode = 'worker';
    if (sessionStorage.getItem(`wms-entry:${workspace.actor_id}`) === 'scan') S.entryMode = 'scan';
    const savedHandoff=sessionStorage.getItem(`wms-handoff:${workspace.actor_id}`);
    if(can('handoff.read')&&savedHandoff&&sessionStorage.getItem(`wms-handoff-view:${workspace.actor_id}`)==='1') {
      S.handoff=(await request(`/api/v1/handoffs/${savedHandoff}?${query()}`)).data;S.handoffID=savedHandoff;S.section='handoffs';S.selected=null;
    }
    render();
  } catch (err) {root.innerHTML = `<main class="login-screen"><h1>Nie udało się otworzyć magazynu</h1><div class="notice error" role="alert">${e(problem(err))}</div><a class="button primary" href="/app/">Zaloguj się ponownie</a><a href="/app/?demo=1">Otwórz magazyn szkoleniowy</a></main>`;}
}
start();
