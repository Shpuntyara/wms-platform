// This fixture is intentionally independent of the primary handoff test helper.
// The old warehouse employee receives broad legacy grants in the bypass cases:
// a denial must prove the custody boundary, rather than a missing API permission.
type criticHandoffFixture struct {
	*fixture
	departments, actors, positions [5]string
}

func newCriticHandoffFixture(t *testing.T) *criticHandoffFixture {
	t.Helper()
	f := newFixture(t)
	if f.scalar(`SELECT (to_regprocedure('warehouse.command_handoff(uuid,uuid,uuid,jsonb)') IS NOT NULL)::text`) != "true" {
		t.Fatal("HANDOFF_FEATURE_UNAVAILABLE: durable two-sided handoff command must exist before behavior checks")
	}
	h := &criticHandoffFixture{fixture: f}
	h.actors[0] = f.actor
	h.positions[0] = f.scalar(`SELECT position_id::text FROM inventory.stock_balances WHERE id=$1`, f.source)
	for n, name := range []string{"Magazyn", "Spawanie", "Szlifowanie", "Lakiernia", "Montaż"} {
		h.departments[n] = id()
		if n > 0 {
			h.actors[n], h.positions[n] = id(), id()
			location := id()
			f.exec(`INSERT INTO platform.actors(id,issuer,subject) VALUES($1,'https://tests.invalid',$1::uuid::text)`, h.actors[n])
			f.exec(`INSERT INTO warehouse.locations(id,organization_id,warehouse_id,code,kind) VALUES($1,$2,$3,$1::uuid::text,'storage')`, location, f.org, f.wh)
			f.exec(`INSERT INTO warehouse.positions(id,organization_id,warehouse_id,location_id) VALUES($1,$2,$3,$4)`, h.positions[n], f.org, f.wh, location)
		}
		kind := "production"
		if n == 0 {
			kind = "warehouse"
		}
		f.exec(`INSERT INTO warehouse.departments(id,organization_id,warehouse_id,code,name,kind,position_id,enabled) VALUES($1,$2,$3,$1::uuid::text,$4,$5,$6,true)`, h.departments[n], f.org, f.wh, name, kind, h.positions[n])
		f.exec(`INSERT INTO warehouse.department_positions(organization_id,warehouse_id,department_id,position_id) VALUES($1,$2,$3,$4)`, f.org, f.wh, h.departments[n], h.positions[n])
		f.exec(`INSERT INTO warehouse.department_employees(actor_id,organization_id,display_name,enabled) VALUES($1,$2,$3,true)`, h.actors[n], f.org, "Pracownik "+name)
		f.exec(`INSERT INTO warehouse.department_members(organization_id,warehouse_id,department_id,actor_id,enabled) VALUES($1,$2,$3,$4,true)`, f.org, f.wh, h.departments[n], h.actors[n])
		for _, permission := range []string{"handoff.send", "handoff.accept", "handoff.read"} {
			f.exec(`INSERT INTO platform.actor_grants(actor_id,organization_id,warehouse_id,owner_id,permission) VALUES($1,$2,$3,$4,$5)`, h.actors[n], f.org, f.wh, f.owner, permission)
		}
		if n > 0 {
			f.exec(`INSERT INTO warehouse.department_routes(organization_id,warehouse_id,source_department_id,destination_department_id,enabled) VALUES($1,$2,$3,$4,true)`, f.org, f.wh, h.departments[n-1], h.departments[n])
		}
	}
	return h
}

func (h *criticHandoffFixture) handlerFor(actor string) http.Handler {
	h.t.Helper()
	registry, err := modules.NewRegistry(nil, nil, nil)
	if err != nil {
		h.t.Fatal(err)
	}
	handler, err := api.New(api.Config{Backend: h.store, Auth: tokenAuth{subject: actor}, Modules: registry})
	if err != nil {
		h.t.Fatal(err)
	}
	return handler
}

func (h *criticHandoffFixture) sendBody(source string, next int, quantity string) map[string]any {
	return map[string]any{"organization_id": h.org, "warehouse_id": h.wh, "source_balance_id": source, "destination_department_id": h.departments[next], "quantity": quantity}
}

func (h *criticHandoffFixture) acceptBody(version int) map[string]any {
	return map[string]any{"organization_id": h.org, "warehouse_id": h.wh, "expected_version": version}
}

func (h *criticHandoffFixture) command(ctx context.Context, command, actor, handoff string, input map[string]any) (json.RawMessage, error) {
	raw, err := json.Marshal(input)
	if err != nil {
		return nil, err
	}
	var result json.RawMessage
	err = h.runtime.QueryRow(ctx, `SELECT warehouse.command_handoff($1::uuid,$2::uuid,$3::uuid,$4::jsonb)`, command, actor, optionalCriticHandoffID(handoff), raw).Scan(&result)
	return result, err
}

func optionalCriticHandoffID(value string) any {
	if value == "" {
		return nil
	}
	return value
}

func criticHandoffObject(t testing.TB, raw json.RawMessage) map[string]any {
	t.Helper()
	var row map[string]any
	if err := json.Unmarshal(raw, &row); err != nil {
		t.Fatalf("HANDOFF_ASSERTION: command did not return an object: %v", err)
	}
	return row
}

func (h *criticHandoffFixture) send(t *testing.T, source string, next int, quantity string) (string, map[string]any) {
	t.Helper()
	raw, err := h.command(t.Context(), id(), h.actors[next-1], "", h.sendBody(source, next, quantity))
	if err != nil {
		t.Fatalf("HANDOFF_ASSERTION: authorized send rejected: %v", err)
	}
	row := criticHandoffObject(t, raw)
	handoff, ok := row["handoff_id"].(string)
	if !ok || handoff == "" {
		t.Fatalf("HANDOFF_ASSERTION: response lacks immutable handoff id: %s", raw)
	}
	return handoff, row
}

func (h *criticHandoffFixture) accept(t *testing.T, handoff string, next int) map[string]any {
	t.Helper()
	raw, err := h.command(t.Context(), id(), h.actors[next], handoff, h.acceptBody(1))
	if err != nil {
		t.Fatalf("HANDOFF_ASSERTION: authenticated recipient rejected: %v", err)
	}
	return criticHandoffObject(t, raw)
}

func (h *criticHandoffFixture) balanceAt(position string) string {
	h.t.Helper()
	return h.scalar(`SELECT id::text FROM inventory.stock_balances WHERE organization_id=$1 AND warehouse_id=$2 AND owner_id=$3 AND item_id=$4 AND position_id=$5 AND stock_state='available' AND lot_id IS NULL AND serial_id IS NULL`, h.org, h.wh, h.owner, h.item, position)
}

func criticHandoffQuantity(t testing.TB, value string) *big.Rat {
	t.Helper()
	result, ok := new(big.Rat).SetString(value)
	if !ok {
		t.Fatalf("HANDOFF_ASSERTION: invalid serialized exact quantity %q", value)
	}
	return result
}

func (h *criticHandoffFixture) assertConservation(t *testing.T, expectedPending *big.Rat) {
	t.Helper()
	physical := h.scalar(`SELECT coalesce(sum(on_hand),0)::text FROM inventory.stock_balances WHERE organization_id=$1 AND warehouse_id=$2 AND owner_id=$3 AND item_id=$4`, h.org, h.wh, h.owner, h.item)
	if criticHandoffQuantity(t, physical).Cmp(big.NewRat(10, 1)) != 0 {
		t.Fatalf("HANDOFF_ASSERTION: transfer changed physical inventory: %s", physical)
	}
	pending := h.scalar(`SELECT coalesce(sum(b.on_hand),0)::text FROM inventory.stock_balances b JOIN warehouse.handoff_transit_positions p ON p.position_id=b.position_id WHERE b.organization_id=$1 AND b.warehouse_id=$2 AND b.owner_id=$3 AND b.item_id=$4`, h.org, h.wh, h.owner, h.item)
	if criticHandoffQuantity(t, pending).Cmp(expectedPending) != 0 {
		t.Fatalf("HANDOFF_ASSERTION: pending custody=%s independent model=%s", pending, expectedPending.RatString())
	}
	if got := h.scalar(`SELECT count(*)::text FROM contracts_v1.stock_balances b JOIN warehouse.handoff_transit_positions p ON p.position_id=b.position_id WHERE b.organization_id=$1 AND (b.available_qty<>0 OR b.reserved_qty<>0 OR b.stock_state<>'in_transit')`, h.org); got != "0" {
		t.Fatalf("HANDOFF_ASSERTION: pending stock is usable, reserved or in the wrong state: %s balances", got)
	}
}

func TestDepartmentHandoffsCriticIndependentConservationHistories(t *testing.T) {
	for seed := uint64(1); seed <= 12; seed++ {
		t.Run(fmt.Sprintf("history_%d", seed), func(t *testing.T) {
			h := newCriticHandoffFixture(t)
			rng := rand.New(rand.NewPCG(0x48414e444f4646, seed))
			// Each parcel has its own identity in the independent oracle. Sending
			// identical stock before accepting in reverse order must not merge custody.
			pending := new(big.Rat)
			parcels := []struct{ handoff, quantity string }{}
			for range 4 {
				quantity := fmt.Sprintf("0.%03d", 1+rng.IntN(999))
				handoff, _ := h.send(t, h.source, 1, quantity)
				pending.Add(pending, criticHandoffQuantity(t, quantity))
				parcels = append(parcels, struct{ handoff, quantity string }{handoff, quantity})
				h.assertConservation(t, pending)
			}
			if got := h.scalar(`SELECT count(DISTINCT transit_position_id)::text FROM warehouse.handoffs WHERE organization_id=$1`, h.org); got != "4" {
				t.Fatalf("HANDOFF_ASSERTION: same-SKU pending parcels share transit identity: %s", got)
			}
			for i := len(parcels) - 1; i >= 0; i-- {
				h.accept(t, parcels[i].handoff, 1)
				pending.Sub(pending, criticHandoffQuantity(t, parcels[i].quantity))
				h.assertConservation(t, pending)
			}
			for next := 2; next < len(h.departments); next++ {
				source := h.balanceAt(h.positions[next-1])
				total := new(big.Rat)
				for _, parcel := range parcels {
					total.Add(total, criticHandoffQuantity(t, parcel.quantity))
				}
				quantity := total.FloatString(3)
				handoff, _ := h.send(t, source, next, quantity)
				h.assertConservation(t, total)
				h.accept(t, handoff, next)
				h.assertConservation(t, new(big.Rat))
			}
		})
	}
}

func TestDepartmentHandoffsCriticConcurrentAcceptIsSingleCustodyChange(t *testing.T) {
	h := newCriticHandoffFixture(t)
	handoff, _ := h.send(t, h.source, 1, "2.125")
	start := make(chan struct{})
	errors := make(chan error, 2)
	var workers sync.WaitGroup
	for range 2 {
		workers.Go(func() {
			<-start
			_, err := h.command(t.Context(), id(), h.actors[1], handoff, h.acceptBody(1))
			errors <- err
		})
	}
	close(start)
	workers.Wait()
	close(errors)
	succeeded := 0
	for err := range errors {
		if err == nil {
			succeeded++
		}
	}
	if succeeded != 1 {
		t.Fatalf("HANDOFF_ASSERTION: concurrent unique-key accepts succeeded %d times, want exactly one", succeeded)
	}
	h.assertConservation(t, new(big.Rat))
	if got := h.scalar(`SELECT version::text||':'||state FROM warehouse.handoffs WHERE id=$1`, handoff); got != "2:accepted" {
		t.Fatalf("HANDOFF_ASSERTION: acceptance did not make one terminal transition: %s", got)
	}
}
